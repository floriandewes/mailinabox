{% trans %}Mail user API (advanced){% endtrans %}

{% trans %}Use your box’s mail user API to add/change/remove users from the command-line or custom services you build.{% endtrans %}

{% trans %}Usage:{% endtrans %}

curl -X VERB [-d "parameters"] --user {email}:{password} https://{{hostname}}/admin/mail/users[action]

{% trans %}Brackets denote an optional argument. Please note that the POST body parameters must be URL-encoded.{% endtrans %}

{% trans %}The email and password given to the --user option must be an administrative user on this system.{% endtrans %}

{% trans %}Verbs{% endtrans %}

{% trans %}Verb{% endtrans %} {% trans %}Action{% endtrans %}
GET{% trans %}(none){% endtrans %} {% trans %}Returns a list of existing mail users. Adding ?format=json to the URL will give JSON-encoded results.{% endtrans %}
POST /add {% trans %}Adds a new mail user. Required POST-body parameters are email and password. Optional parameters: privilege=admin and quota{% endtrans %}
POST /remove {% trans %}Removes a mail user. Required POST-by parameter is email.{% endtrans %}
POST/privileges/add {% trans %}Used to make a mail user an admin. Required POST-body parameters are email and privilege=admin.{% endtrans %}
POST/privileges/remove {% trans %}Used to remove the admin privilege from a mail user. Required POST-body parameter is email.{% endtrans %}
GET /quota {% trans %}Get the quota for a mail user. Required POST-body parameters are email and will return JSON result{% endtrans %}
POST /quota {% trans %}Set the quota for a mail user. Required POST-body parameters are email and quota.{% endtrans %}

{% trans %}Examples:{% endtrans %}

{% trans %}Try these examples. For simplicity the examples omit the --user me@mydomain.com:yourpassword command line argument which you must fill in with your administrative email address and password.{% endtrans %}

# {% trans %}Gives a JSON-encoded list of all mail users{% endtrans %}
curl -X GET https://{{hostname}}/admin/mail/users?format=json

# {% trans %}Adds a new email user{% endtrans %}
curl -X POST -d "email=new_user@mydomail.com" -d "password=s3curE_pa5Sw0rD" https://{{hostname}}/admin/mail/users/add

# {% trans %}Removes a email user{% endtrans %}
curl -X POST -d "email=new_user@mydomail.com" https://{{hostname}}/admin/mail/users/remove

# {% trans %}Adds admin privilege to an email user{% endtrans %}
curl -X POST -d "email=new_user@mydomail.com" -d "privilege=admin" https://{{hostname}}/admin/mail/users/privileges/add

# {% trans %}Removes admin privilege from an email user{% endtrans %}
curl -X POST -d "email=new_user@mydomail.com" https://{{hostname}}/admin/mail/users/privileges/remove
